41. Top N per Group: Find the top 2 most expensive products (by original_price) for vendor_id 7.
-> select vendor_id,product_id,original_price from vendor_inventory
having vendor_id = 7
order by original_price desc
limit 2

42. Customer Loyalty (Frequency): Identify customers who have made purchases on more than 5 distinct dates.
-> select distinct customer_id,count(distinct market_date) "Purchase_Date_count" from customer_purchases 
group by customer_id
having count(distinct market_date) > 5

45. Basket Size Analysis: Calculate the average number of items (quantity) purchased per single transaction (defined by a unique combination of customer, vendor, and time).
-> select avg(a.Average_Qty) as "Basket_Size" from 
(select customer_id,vendor_id,sum(quantity) as "Average_Qty",transaction_time from customer_purchases
group by customer_id,vendor_id,transaction_time) as a;

46. Relative Performance: Calculate a "Performance Score" for each product: (Total Qty Sold of Product) / (Total Qty Sold of All Products in that Category).
-> 
select p.product_id,p.product_category_id,pc.product_category_name, sum(cp.quantity) as "product_total",
(sum(cp.quantity)*1.0/sum(sum(cp.quantity)) over(partition by pc.product_category_id))*100 as "Performance_score"

from product as p
join customer_purchases as cp
on p.product_id = cp.product_id
join product_category as pc
on pc.product_category_id = p.product_category_id

group by p.product_id,p.product_category_id,pc.product_category_name
