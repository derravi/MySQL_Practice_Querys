✅ HARD Answers (7)
✅ H1 (Revenue per date > 500)
SELECT market_date,
       ROUND(SUM(quantity * cost_to_customer_per_qty), 2) AS total_revenue
FROM customer_purchases
GROUP BY market_date
HAVING total_revenue > 500
ORDER BY total_revenue DESC;

✅ H2 (Top 3 products by revenue)
SELECT product_id,
       ROUND(SUM(quantity * cost_to_customer_per_qty), 2) AS total_revenue
FROM customer_purchases
GROUP BY product_id
ORDER BY total_revenue DESC
LIMIT 3;

✅ H3 (Vendor revenue + total quantity sold)
SELECT vendor_id,
       SUM(quantity) AS total_quantity_sold,
       ROUND(SUM(quantity * cost_to_customer_per_qty), 2) AS total_revenue
FROM customer_purchases
GROUP BY vendor_id
ORDER BY total_revenue DESC;

✅ H4 (Unique customers per date)
SELECT market_date,
       COUNT(DISTINCT customer_id) AS unique_customers
FROM customer_purchases
GROUP BY market_date
ORDER BY unique_customers DESC;

✅ H5 (Product with highest quantity sold)
SELECT product_id,
       SUM(quantity) AS total_quantity_sold
FROM customer_purchases
GROUP BY product_id
ORDER BY total_quantity_sold DESC
LIMIT 1;

✅ H6 (Revenue per market_year using join)
SELECT mdi.market_year,
       ROUND(SUM(cp.quantity * cp.cost_to_customer_per_qty), 2) AS total_revenue
FROM market_date_info mdi
JOIN customer_purchases cp
ON mdi.market_date = cp.market_date
GROUP BY mdi.market_year
ORDER BY total_revenue DESC;

✅ H7 (Vendor revenue + unique customers served)
SELECT vendor_id,
       ROUND(SUM(quantity * cost_to_customer_per_qty), 2) AS total_revenue,
       COUNT(DISTINCT customer_id) AS unique_customers_served
FROM customer_purchases
GROUP BY vendor_id
ORDER BY total_revenue DESC;
